// Sketch for frontend/src/components/map/useMapMotion.ts (not wired into the repo).
// Owns the map's motion policy so no component reads the clock, the battery or
// the reduced-motion flag itself. React never runs in the frame loop: the hook
// hands out imperative helpers; per-tick work is MapLibre paint calls only.
//
// Tokens mirror lib/motion.ts of the engineering notes (ms): instant 90,
// fast 160, base 260, slow 360. Easings are CSS cubic-bezier control points.

import type { Map as MlMap, PaddingOptions } from "maplibre-gl";
import { useCallback, useEffect, useMemo, useRef, useState } from "react";

export const dur = { instant: 90, fast: 160, base: 260, slow: 360 } as const;
export const bezier = {
  out: [0.2, 0.8, 0.2, 1],
  inOut: [0.4, 0, 0.2, 1],
  in: [0.4, 0, 1, 1],
} as const;

/** Cubic-bezier solver (MapLibre camera easing takes a function t -> t). */
export function cubicBezier([x1, y1, x2, y2]: readonly [number, number, number, number]) {
  const cx = 3 * x1, bx = 3 * (x2 - x1) - cx, ax = 1 - cx - bx;
  const cy = 3 * y1, by = 3 * (y2 - y1) - cy, ay = 1 - cy - by;
  const X = (t: number) => ((ax * t + bx) * t + cx) * t;
  const Y = (t: number) => ((ay * t + by) * t + cy) * t;
  return (x: number) => {
    let t = x;
    for (let i = 0; i < 8; i++) {
      const err = X(t) - x;
      const d = (3 * ax * t + 2 * bx) * t + cx;
      if (Math.abs(err) < 1e-5 || d === 0) break;
      t -= err / d;
    }
    return Y(t);
  };
}
const easeOut = cubicBezier(bezier.out);

export type Mode = "full" | "reduced";

/** Why ambient (looping) motion is paused. Empty set means it may run. */
export type PauseReason = "hidden" | "idle" | "reduced" | "battery" | "offscreen";

const IDLE_MS = 20_000; // no pointer/touch/camera input for this long -> stop ambient
const LOW_BATTERY = 0.2;

interface BatteryManager extends EventTarget {
  level: number;
  charging: boolean;
}

export interface MapMotion {
  mode: Mode;
  /** True when looping motion (flow, ripple) is allowed right now. */
  ambient: boolean;
  pauseReasons: ReadonlySet<PauseReason>;
  /** Camera move to a selection; duration 0 in reduced mode. Pass sheet padding. */
  easeToSelection(opts: {
    center: [number, number];
    zoom?: number;
    padding?: PaddingOptions;
    ms?: number;
  }): void;
  /** Paint-property transition (colour, opacity) set by the map, not by JS ticks. */
  transition(layerId: string, props: string[], ms?: number): void;
  /** Register a looping effect. It is started and stopped by the policy. */
  ambientLoop(fps: number, tick: (i: number) => void): () => void;
  /** Call from layers that animate once (draw-in); returns the duration to use. */
  oneShotMs(ms: number): number;
}

export function useMapMotion(map: MlMap | null, mode: Mode): MapMotion {
  const [reasons, setReasons] = useState<ReadonlySet<PauseReason>>(new Set());
  const loops = useRef(new Set<{ fps: number; tick: (i: number) => void; i: number; last: number }>());
  const raf = useRef(0);

  const setReason = useCallback((r: PauseReason, on: boolean) => {
    setReasons((prev) => {
      if (prev.has(r) === on) return prev;
      const next = new Set(prev);
      if (on) next.add(r);
      else next.delete(r);
      return next;
    });
  }, []);

  useEffect(() => setReason("reduced", mode === "reduced"), [mode, setReason]);

  useEffect(() => {
    const onVis = () => setReason("hidden", document.hidden);
    onVis();
    document.addEventListener("visibilitychange", onVis);
    return () => document.removeEventListener("visibilitychange", onVis);
  }, [setReason]);

  useEffect(() => {
    let timer: ReturnType<typeof setTimeout>;
    const wake = () => {
      setReason("idle", false);
      clearTimeout(timer);
      timer = setTimeout(() => setReason("idle", true), IDLE_MS);
    };
    wake();
    const opts = { passive: true } as const;
    const evs = ["pointerdown", "pointermove", "keydown", "wheel", "touchstart"] as const;
    for (const e of evs) window.addEventListener(e, wake, opts);
    map?.on("movestart", wake);
    return () => {
      clearTimeout(timer);
      for (const e of evs) window.removeEventListener(e, wake);
      map?.off("movestart", wake);
    };
  }, [map, setReason]);

  useEffect(() => {
    // Battery API exists on Chromium only; absence means no battery rule.
    const nav = navigator as Navigator & { getBattery?: () => Promise<BatteryManager> };
    if (!nav.getBattery) return;
    let bm: BatteryManager | undefined;
    const check = () => bm && setReason("battery", !bm.charging && bm.level <= LOW_BATTERY);
    let dead = false;
    nav.getBattery().then((b) => {
      if (dead) return;
      bm = b;
      check();
      b.addEventListener("levelchange", check);
      b.addEventListener("chargingchange", check);
    });
    return () => {
      dead = true;
      bm?.removeEventListener("levelchange", check);
      bm?.removeEventListener("chargingchange", check);
    };
  }, [setReason]);

  // The only rAF in the feature: one shared clock for every ambient loop.
  // It runs only while no pause reason is set and at least one loop exists.
  const ambient = reasons.size === 0;
  const pump = useRef<() => void>(() => {});
  useEffect(() => {
    if (!ambient || !map) {
      pump.current = () => {};
      return;
    }
    const frame = (ts: number) => {
      raf.current = 0;
      for (const l of loops.current) {
        if (ts - l.last >= 1000 / l.fps - 1) {
          l.last = ts;
          l.tick(l.i++);
        }
      }
      if (loops.current.size) raf.current = requestAnimationFrame(frame);
    };
    pump.current = () => {
      if (!raf.current) raf.current = requestAnimationFrame(frame);
    };
    pump.current();
    return () => {
      cancelAnimationFrame(raf.current);
      raf.current = 0;
    };
  }, [ambient, map]);

  return useMemo<MapMotion>(() => {
    const reduced = mode === "reduced";
    return {
      mode,
      ambient,
      pauseReasons: reasons,
      easeToSelection({ center, zoom, padding, ms = dur.base }) {
        // essential:true so MapLibre's own OS check does not override the
        // in-app setting; reduced mode is expressed as duration 0 instead.
        map?.easeTo({ center, zoom, padding, duration: reduced ? 0 : ms, easing: easeOut, essential: true });
      },
      transition(layerId, props, ms = dur.base) {
        for (const p of props) map?.setPaintProperty(layerId, `${p}-transition`, { duration: reduced ? 0 : ms, delay: 0 });
      },
      ambientLoop(fps, tick) {
        const l = { fps, tick, i: 0, last: 0 };
        loops.current.add(l);
        pump.current();
        return () => loops.current.delete(l);
      },
      oneShotMs: (ms) => (reduced ? 0 : ms),
    };
  }, [map, mode, ambient, reasons]);
}
